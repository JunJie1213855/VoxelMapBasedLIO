cmake_minimum_required(VERSION 3.16)
project(pv_lio LANGUAGES C CXX)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL 1.10 REQUIRED COMPONENTS common filters io kdtree)
find_package(OpenMP REQUIRED)
find_package(Threads REQUIRED)

set(VOXELMAP_OMP_THREADS 4 CACHE STRING "Number of OpenMP workers used by voxel mapping")
set(ros_dependencies rclcpp builtin_interfaces std_msgs sensor_msgs geometry_msgs
  nav_msgs visualization_msgs tf2 tf2_ros pcl_conversions livox_ros_driver2)

add_executable(pv_lio_node src/voxelMapping.cpp src/preprocess.cpp)
target_include_directories(pv_lio_node PRIVATE include ${PCL_INCLUDE_DIRS})
target_compile_definitions(pv_lio_node PRIVATE
  ROOT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/"
  MP_EN MP_PROC_NUM=${VOXELMAP_OMP_THREADS})
ament_target_dependencies(pv_lio_node ${ros_dependencies})
target_link_libraries(pv_lio_node Eigen3::Eigen ${PCL_LIBRARIES} OpenMP::OpenMP_CXX)

foreach(converter rs_to_velodyne pandar_to_velodyne)
  add_executable(${converter} src/converter/${converter}.cpp)
  target_include_directories(${converter} PRIVATE include ${PCL_INCLUDE_DIRS})
  ament_target_dependencies(${converter} rclcpp sensor_msgs geometry_msgs tf2 tf2_ros pcl_conversions)
  target_link_libraries(${converter} ${PCL_LIBRARIES})
endforeach()

install(TARGETS pv_lio_node rs_to_velodyne pandar_to_velodyne
  DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README_ROS2.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(async_worker_test test/async_worker_test.cpp)
  target_include_directories(async_worker_test PRIVATE include)
  target_link_libraries(async_worker_test Threads::Threads)
  ament_add_gtest(display_map_test test/display_map_test.cpp)
  target_include_directories(display_map_test PRIVATE include ${PCL_INCLUDE_DIRS})
  target_link_libraries(display_map_test Eigen3::Eigen ${PCL_LIBRARIES})
  ament_add_gtest(preprocess_test test/preprocess_test.cpp src/preprocess.cpp)
  target_include_directories(preprocess_test PRIVATE include src ${PCL_INCLUDE_DIRS})
  ament_target_dependencies(preprocess_test ${ros_dependencies})
  target_link_libraries(preprocess_test Eigen3::Eigen ${PCL_LIBRARIES} OpenMP::OpenMP_CXX)
endif()

ament_package()
