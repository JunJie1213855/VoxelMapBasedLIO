cmake_minimum_required(VERSION 3.16)
project(voxel_map LANGUAGES C CXX)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL 1.10 REQUIRED COMPONENTS common filters io kdtree)
find_package(OpenMP REQUIRED)
find_package(Threads REQUIRED)

set(VOXELMAP_OMP_THREADS 4 CACHE STRING "Number of OpenMP workers used by voxel mapping")
set(ros_dependencies rclcpp builtin_interfaces std_msgs sensor_msgs geometry_msgs
  nav_msgs visualization_msgs tf2 tf2_ros pcl_conversions livox_ros_driver2)

find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/Pose6D.msg" "msg/States.msg"
  DEPENDENCIES std_msgs)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)

add_executable(voxel_mapping_odom src/voxelMapping.cpp src/preprocess.cpp)
target_include_directories(voxel_mapping_odom PRIVATE include ${PCL_INCLUDE_DIRS})
target_compile_definitions(voxel_mapping_odom PRIVATE
  ROOT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/"
  MP_EN MP_PROC_NUM=${VOXELMAP_OMP_THREADS})
ament_target_dependencies(voxel_mapping_odom ${ros_dependencies})
target_link_libraries(voxel_mapping_odom Eigen3::Eigen ${PCL_LIBRARIES} OpenMP::OpenMP_CXX Threads::Threads "${cpp_typesupport_target}")

install(TARGETS voxel_mapping_odom
  DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz_cfg DESTINATION share/${PROJECT_NAME})
install(FILES README_ROS2.md DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(robosense_preprocess_test test/robosense_preprocess_test.cpp src/preprocess.cpp)
  target_include_directories(robosense_preprocess_test PRIVATE include src ${PCL_INCLUDE_DIRS})
  ament_target_dependencies(robosense_preprocess_test ${ros_dependencies})
  target_link_libraries(robosense_preprocess_test Eigen3::Eigen ${PCL_LIBRARIES})
  ament_add_gtest(imu_process_test test/imu_process_test.cpp)
  target_include_directories(imu_process_test PRIVATE include src ${PCL_INCLUDE_DIRS})
  ament_target_dependencies(imu_process_test ${ros_dependencies})
  target_link_libraries(imu_process_test Eigen3::Eigen ${PCL_LIBRARIES} "${cpp_typesupport_target}")
  ament_add_gtest(plane_test test/plane_test.cpp)
  target_include_directories(plane_test PRIVATE include ${PCL_INCLUDE_DIRS})
  ament_target_dependencies(plane_test ${ros_dependencies})
  target_link_libraries(plane_test Eigen3::Eigen ${PCL_LIBRARIES} OpenMP::OpenMP_CXX Threads::Threads "${cpp_typesupport_target}")
  ament_add_gtest(display_map_test test/display_map_test.cpp)
  target_include_directories(display_map_test PRIVATE include ${PCL_INCLUDE_DIRS})
  target_link_libraries(display_map_test ${PCL_LIBRARIES})
  ament_add_gtest(async_worker_test test/async_worker_test.cpp)
  target_include_directories(async_worker_test PRIVATE include)
  target_link_libraries(async_worker_test Threads::Threads)
endif()
ament_package()
